🔗Lookup - Cruzamento de Coleções

🎯 Buscar alunos COM dados do curso
db["alunos"].aggregate([
  {
    $lookup: {
      from: "cursos",           // coleção para fazer join
      localField: "curso_id",   // campo local (alunos)
      foreignField: "_id",      // campo da coleção externa (cursos)
      as: "dados_curso"         // nome do array resultado
    }
  },
  {
    $unwind: "$dados_curso"     // transforma array em objeto
  },
  {
    $project: {
      nome: 1,
      idade: 1,
      "curso_nome": "$dados_curso.nome",
      "departamento": "$dados_curso.departamento",
      "coordenador": "$dados_curso.coordenador"
    }
  }
])
🎯 Lookup com múltiplas coleções (aluno + curso + turma)
db["alunos"].aggregate([
  // Join com cursos
  {
    $lookup: {
      from: "cursos",
      localField: "curso_id",
      foreignField: "_id",
      as: "curso"
    }
  },
  // Join com turmas
  {
    $lookup: {
      from: "turmas",
      localField: "turma_id",
      foreignField: "_id",
      as: "turma"
    }
  },
  // Transformar arrays em objetos
  { $unwind: "$curso" },
  { $unwind: "$turma" },
  // Projeção final
  {
    $project: {
      nome: 1,
      idade: 1,
      "curso_nome": "$curso.nome",
      "turma_codigo": "$turma.codigo",
      "semestre": "$turma.semestre",
      media: { $avg: "$notas" }
    }
  }
])

🧠 Conceitos importantes:

  • $lookup: faz "JOIN" entre coleções (como SQL)
  • $unwind: converte array em objetos separados
  • $project: seleciona e renomeia campos
  • Pipeline: sequência de operações (como um funil)

def buscar_dados_completos(aluno_id):
    # Banco escola
    aluno = db_escola["alunos"].find_one({"_id": aluno_id})
    curso = db_escola["cursos"].find_one({"_id": aluno["curso_id"]})

    # Banco biblioteca
    emprestimos = list(db_biblioteca["emprestimos"].find({"aluno_id": aluno_id}))

    # Banco financeiro
    client_financeiro = MongoClient("mongodb://servidor2:27017")
    db_financeiro = client_financeiro["financeiro"]
    mensalidades = list(db_financeiro["pagamentos"].find({"aluno_id": aluno_id}))

    # Combinar tudo
    return {
        "aluno": aluno,
        "curso": curso,
        "emprestimos": emprestimos,
        "situacao_financeira": mensalidades
    }

📈Ordenação ASC/DESC e Limitação

⬆️⬇️ Ordenação Simples

Crescente (ASC) = 1
// Por idade (menor para maior)
db.alunos.find().sort({ idade: 1 })

// Por nome (A para Z)
db.alunos.find().sort({ nome: 1 })
Decrescente (DESC) = -1
// Por idade (maior para menor)
db.alunos.find().sort({ idade: -1 })

// Por data (mais recente primeiro)
db.alunos.find().sort({
  data_matricula: -1
})

🎯 Ordenação Múltipla

Múltiplos campos
// 1º por curso, depois por idade
db.alunos.find().sort({
  curso_id: 1,
  idade: -1
})

// 1º por ativo, 2º por nota média
db.alunos.aggregate([
  {
    $addFields: {
      media: { $avg: "$notas" }
    }
  },
  {
    $sort: {
      ativo: -1,    // ativos primeiro
      media: -1     // melhores notas primeiro
    }
  }
])
📋 Limitação e Paginação
// Apenas os 5 primeiros
db.alunos.find().limit(5)

// Pular 10 e pegar os próximos 5 (página 3, tamanho 5)
db.alunos.find().skip(10).limit(5)

// Top 3 alunos por média (pipeline)
db.alunos.aggregate([
  {
    $addFields: {
      media: { $avg: "$notas" }
    }
  },
  {
    $sort: { media: -1 }
  },
  {
    $limit: 3
  },
  {
    $project: {
      nome: 1,
      media: 1,
      _id: 0
    }
  }
])

⚡ Dicas de Performance:

  • Índices: Crie índices nos campos de ordenação para speed
  • Limit sempre: Use limit() para evitar resultados gigantes
  • Pipeline otimizado: $match antes de $sort quando possível

🧮Agregações e Pipeline Avançado

🧮 Classificar alunos (aprovados/reprovados)
db.alunos.aggregate([
  {
    $addFields: {
      status: {
        $cond: {
          if: { $gte: [{ $avg: "$notas" }, 7] },
          then: "Aprovado",
          else: "Reprovado"
        }
      }
    }
  },
  {
    $group: {
      _id: "$status",
      count: { $sum: 1 }
    }
  }
])
📊 Estatísticas por Curso
db.alunos.aggregate([
  // Filtrar apenas alunos ativos
  { $match: { ativo: true } },

  // Calcular média individual
  {
    $addFields: {
      media_individual: { $avg: "$notas" }
    }
  },

  // Agrupar por curso
  {
    $group: {
      _id: "$curso_id",
      total_alunos: { $sum: 1 },
      idade_media: { $avg: "$idade" },
      media_geral: { $avg: "$media_individual" },
      melhor_nota: { $max: "$media_individual" },
      pior_nota: { $min: "$media_individual" }
    }
  },

  // Adicionar dados do curso
  {
    $lookup: {
      from: "cursos",
      localField: "_id",
      foreignField: "_id",
      as: "curso_info"
    }
  },

  // Estruturar resultado final
  {
    $project: {
      curso_nome: { $arrayElemAt: ["$curso_info.nome", 0] },
      total_alunos: 1,
      idade_media: { $round: ["$idade_media", 1] },
      media_geral: { $round: ["$media_geral", 2] },
      melhor_nota: { $round: ["$melhor_nota", 2] },
      pior_nota: { $round: ["$pior_nota", 2] }
    }
  },

  // Ordenar por média geral
  { $sort: { media_geral: -1 } }
])
📈 Contagem e Agrupamento por Faixa Etária
db.alunos.aggregate([
  {
    $bucket: {
      groupBy: "$idade",
      boundaries: [18, 21, 25, 30],
      default: "30+",
      output: {
        count: { $sum: 1 },
        nomes: { $push: "$nome" }
      }
    }
  }
])

🔥 Operadores de Agregação Úteis:

  • $match: filtrar documentos (como WHERE)
  • $group: agrupar e calcular (como GROUP BY)
  • $project: selecionar campos (como SELECT)
  • $sort: ordenar resultados
  • $limit/$skip: paginação
  • $lookup: JOIN entre coleções
  • $addFields: adicionar campos calculados
  • $unwind: desestruturar arrays
  • $bucket: agrupar por faixas de valores

🚀 Classe Completa para Gestão Escolar
from pymongo import MongoClient, ASCENDING, DESCENDING
from bson import ObjectId
from datetime import datetime, timedelta
import re

class GestaoEscolar:
    def __init__(self, connection_string="mongodb://localhost:27017", db_name="escola"):
        self.client = MongoClient(connection_string)
        self.db = self.client[db_name]
        self.alunos = self.db["alunos"]
        self.cursos = self.db["cursos"]
        self.turmas = self.db["turmas"]

        # Criar índices automaticamente
        self._criar_indices()

    def _criar_indices(self):
        """Cria índices essenciais para performance"""
        try:
            # Índices para alunos
            self.alunos.create_index([("nome", 1)])
            self.alunos.create_index([("curso_id", 1), ("ativo", 1)])
            self.alunos.create_index([("data_matricula", -1)])
            self.alunos.create_index([("nome", "text")])

            # Índices para cursos
            self.cursos.create_index([("nome", 1)])
            self.cursos.create_index([("departamento", 1)])

            print("✅ Índices criados com sucesso!")
        except Exception as e:
            print(f"⚠️ Erro ao criar índices: {e}")

    def buscar_alunos_completo(self, filtros=None, ordenacao=None, limite=None, pagina=1):
        """
        Busca avançada com filtros, ordenação e paginação
        """
        pipeline = []

        # Aplicar filtros
        if filtros:
            pipeline.append({"$match": filtros})

        # Lookup com curso
        pipeline.extend([
            {
                "$lookup": {
                    "from": "cursos",
                    "localField": "curso_id",
                    "foreignField": "_id",
                    "as": "curso"
                }
            },
            {"$unwind": {"path": "$curso", "preserveNullAndEmptyArrays": True}},

            # Adicionar campos calculados
            {
                "$addFields": {
                    "media": {"$avg": "$notas"},
                    "total_notas": {"$size": {"$ifNull": ["$notas", []]}},
                    "curso_nome": {"$ifNull": ["$curso.nome", "Sem curso"]},
                    "departamento": {"$ifNull": ["$curso.departamento", "Sem departamento"]}
                }
            }
        ])

        # Aplicar ordenação
        if ordenacao:
            pipeline.append({"$sort": ordenacao})
        else:
            pipeline.append({"$sort": {"nome": 1}})

        # Paginação
        if limite:
            skip = (pagina - 1) * limite
            pipeline.extend([
                {"$skip": skip},
                {"$limit": limite}
            ])

        return list(self.alunos.aggregate(pipeline))

    def estatisticas_gerais(self):
        """Gera relatório completo de estatísticas"""
        pipeline = [
            {
                "$facet": {
                    # Estatísticas por curso
                    "por_curso": [
                        {"$match": {"ativo": True}},
                        {
                            "$group": {
                                "_id": "$curso_id",
                                "total_alunos": {"$sum": 1},
                                "media_idade": {"$avg": "$idade"},
                                "media_notas": {"$avg": {"$avg": "$notas"}}
                            }
                        },
                        {
                            "$lookup": {
                                "from": "cursos",
                                "localField": "_id",
                                "foreignField": "_id",
                                "as": "curso"
                            }
                        },
                        {"$unwind": "$curso"},
                        {
                            "$project": {
                                "curso_nome": "$curso.nome",
                                "departamento": "$curso.departamento",
                                "total_alunos": 1,
                                "media_idade": {"$round": ["$media_idade", 1]},
                                "media_notas": {"$round": ["$media_notas", 2]}
                            }
                        }
                    ],

                    # Distribuição por faixa etária
                    "por_idade": [
                        {
                            "$bucket": {
                                "groupBy": "$idade",
                                "boundaries": [16, 20, 25, 30, 40],
                                "default": "40+",
                                "output": {
                                    "count": {"$sum": 1},
                                    "media_notas": {"$avg": {"$avg": "$notas"}}
                                }
                            }
                        }
                    ],

                    # Alunos por status
                    "por_status": [
                        {
                            "$addFields": {
                                "status": {
                                    "$cond": {
                                        "if": {"$gte": [{"$avg": "$notas"}, 7]},
                                        "then": "Aprovado",
                                        "else": "Reprovado"
                                    }
                                }
                            }
                        },
                        {
                            "$group": {
                                "_id": "$status",
                                "count": {"$sum": 1}
                            }
                        }
                    ]
                }
            }
        ]

        resultado = list(self.alunos.aggregate(pipeline))[0]
        return resultado

    def busca_textual_avancada(self, termo_busca, incluir_curso=True):
        """Busca textual avançada com score de relevância"""
        pipeline = [
            {"$match": {"$text": {"$search": termo_busca}}},
            {"$addFields": {"score": {"$meta": "textScore"}}},
        ]

        if incluir_curso:
            pipeline.extend([
                {
                    "$lookup": {
                        "from": "cursos",
                        "localField": "curso_id",
                        "foreignField": "_id",
                        "as": "curso"
                    }
                },
                {"$unwind": {"path": "$curso", "preserveNullAndEmptyArrays": True}}
            ])

        pipeline.extend([
            {"$sort": {"score": {"$meta": "textScore"}, "nome": 1}},
            {
                "$project": {
                    "nome": 1,
                    "idade": 1,
                    "curso_nome": {"$ifNull": ["$curso.nome", "Sem curso"]},
                    "media": {"$avg": "$notas"},
                    "score": 1
                }
            }
        ])

        return list(self.alunos.aggregate(pipeline))

    def relatorio_mensal(self, ano=None, mes=None):
        """Relatório de matrículas por mês"""
        if not ano:
            ano = datetime.now().year
        if not mes:
            mes = datetime.now().month

        inicio_mes = datetime(ano, mes, 1)
        if mes == 12:
            fim_mes = datetime(ano + 1, 1, 1)
        else:
            fim_mes = datetime(ano, mes + 1, 1)

        pipeline = [
            {
                "$match": {
                    "data_matricula": {
                        "$gte": inicio_mes,
                        "$lt": fim_mes
                    }
                }
            },
            {
                "$lookup": {
                    "from": "cursos",
                    "localField": "curso_id",
                    "foreignField": "_id",
                    "as": "curso"
                }
            },
            {"$unwind": "$curso"},
            {
                "$group": {
                    "_id": {
                        "curso": "$curso.nome",
                        "dia": {"$dayOfMonth": "$data_matricula"}
                    },
                    "count": {"$sum": 1},
                    "alunos": {"$push": "$nome"}
                }
            },
            {"$sort": {"_id.dia": 1, "_id.curso": 1}}
        ]

        return list(self.alunos.aggregate(pipeline))

🤔Por que Índices Existem?

Pense no índice de um livro: sem ele, achar um assunto significa ler página por página. Com ele, você vai direto na página certa.

No MongoDB é a mesma ideia: sem índice, toda consulta faz o banco examinar documento por documento — isso se chama COLLSCAN (collection scan).

📥 Preparando o cenário: uma coleção grande
// Popular uma coleção separada com 50 mil documentos (execute uma vez)
const bulk = []
for (let i = 0; i < 50000; i++) {
  bulk.push({
    nome: "Aluno " + i,
    idade: 18 + Math.floor(Math.random() * 50),   // 18 a 67
    ativo: Math.random() > 0.2,
    notas: [Math.random() * 10, Math.random() * 10]
  })
}
db.alunos_grande.insertMany(bulk)

totalKeysExamined 0 ~987
executionTimeMillis ~38 ms ~1 ms

🔑 Como ler o explain():

  • stage: COLLSCAN = varreu a coleção inteira; IXSCAN = usou um índice
  • totalDocsExamined: quantos documentos o banco teve que abrir e olhar
  • nReturned: quantos documentos realmente satisfazem o filtro (o alvo)
  • Regra de ouro: quanto mais perto totalDocsExamined fica de nReturned, melhor o índice está sendo aproveitado

🗂️Tipos de Índice e Quando Usar

📈 Criando Índices Essenciais
// Índice simples (campo único)
db.alunos.createIndex({ nome: 1 })
db.alunos.createIndex({ idade: -1 })
db.alunos.createIndex({ data_matricula: -1 })

// Índice composto (múltiplos campos)
db.alunos.createIndex({ curso_id: 1, ativo: 1 })
db.alunos.createIndex({ ativo: 1, idade: -1 })

// Índice de texto para busca textual
db.alunos.createIndex({ nome: "text", curso: "text" })

// Índice esparso (só para docs que têm o campo)
db.alunos.createIndex({ telefone: 1 }, { sparse: true })

// Índice com TTL (expira documentos automaticamente)
db.logs.createIndex({ data_criacao: 1 }, { expireAfterSeconds: 86400 }) // 24h

// Índice único (garante unicidade)
db.alunos.createIndex({ email: 1 }, { unique: true })

📐 Regra ESR para índices compostos (Equality → Sort → Range)

Quando o índice tem mais de um campo, a ordem importa. A regra ESR diz: primeiro os campos de igualdade (Equality), depois os de ordenação (Sort), por último os de intervalo (Range).

// Consulta: alunos ativos, ordenados por nome, com idade > 25
db.alunos.find(
  { ativo: true, idade: { $gt: 25 } }
).sort({ nome: 1 })

// ✅ ESR: Equality(ativo) → Sort(nome) → Range(idade)
db.alunos.createIndex({ ativo: 1, nome: 1, idade: 1 })

// ❌ Ordem errada: Range antes do Sort — MongoDB não consegue
// usar o índice para ordenar e faz um sort em memória (mais lento)
db.alunos.createIndex({ idade: 1, ativo: 1, nome: 1 })

🎯 Estratégias de Indexação:

✅ Fazer

  • Indexar campos de consulta frequente
  • Usar índices compostos para consultas múltiplas (regra ESR)
  • Monitorar com explain() regularmente
  • Criar índices em background em produção

❌ Evitar

  • Muitos índices (impacta writes, como visto acima)
  • Índices em campos que mudam muito
  • Índices em coleções muito pequenas
  • Índices duplicados ou redundantes

4. Índice composto com ESR
// Equality → produto (igualdade)
// Sort     → data (ordenação)

Essa ordem é correta porque o MongoDB primeiro filtra por igualdade (produto), depois já entrega os resultados na ordem pedida pelo sort (data) sem precisar ordenar em memória, e só então aplica o filtro de intervalo (valor) dentro desse subconjunto.

                turma_id: novaTurma._id,

        { $inc: { vagas_ocupadas: 1 } },

🔄 Função Reutilizável para Transferências
// Função que pode ser reutilizada com diferentes alunos/turmas
function transferirAluno(nomeAluno, codigoNovaTurma) {
    const session = db.getMongo().startSession()

    try {
        session.startTransaction()

        // Buscar aluno
        const aluno = db.alunos.findOne(
            { nome: nomeAluno, ativo: true },
            { session: session }
        )

        if (!aluno) {
            throw new Error(`Aluno "${nomeAluno}" não encontrado ou inativo`)
        }

        // Buscar nova turma
        const novaTurma = db.turmas.findOne(
            { codigo: codigoNovaTurma },
            { session: session }
        )

        if (!novaTurma) {
            throw new Error(`Turma "${codigoNovaTurma}" não encontrada`)
        }

        // Verificar se não é a mesma turma
        if (aluno.turma_id.equals(novaTurma._id)) {
            throw new Error("Aluno já está nesta turma")
        }

        // Verificar vagas
        if (novaTurma.vagas_ocupadas >= novaTurma.vagas) {
            throw new Error(`Turma ${codigoNovaTurma} está lotada`)
        }

        // Executar transferência
        db.alunos.updateOne(
            { _id: aluno._id },
            {
                $set: {
                    turma_id: novaTurma._id,
                    curso_id: novaTurma.curso_id,
                    data_transferencia: new Date()
                }
            },
            { session: session }
        )

        // Atualizar vagas
        db.turmas.updateOne(
            { _id: aluno.turma_id },
            { $inc: { vagas_ocupadas: -1 } },
            { session: session }
        )

        db.turmas.updateOne(
            { _id: novaTurma._id },
            { $inc: { vagas_ocupadas: 1 } },
            { session: session }
        )

        session.commitTransaction()
        return { success: true, message: "Transferência realizada com sucesso!" }

    } catch (error) {
        session.abortTransaction()
        return { success: false, message: error.message }
    } finally {
        session.endSession()
    }
}

from pymongo import MongoClient
from pymongo.errors import PyMongoError

def transferir_aluno(aluno_id, nova_turma_id):
    client = MongoClient()

    with client.start_session() as session:
        try:
            with session.start_transaction():
                # Buscar dados atuais
                aluno = client.escola.alunos.find_one(
                    {"_id": aluno_id}, session=session
                )

                if not aluno:
                    raise ValueError("Aluno não encontrado")

                turma_atual = aluno.get("turma_id")

                # Remover da turma atual
                if turma_atual:
                    client.escola.turmas.update_one(
                        {"_id": turma_atual},
                        {"$pull": {"alunos": aluno_id}},
                        session=session
                    )

                # Adicionar na nova turma
                client.escola.turmas.update_one(
                    {"_id": nova_turma_id},
                    {"$push": {"alunos": aluno_id}},
                    session=session
                )

                # Atualizar aluno
                client.escola.alunos.update_one(
                    {"_id": aluno_id},
                    {
                        "$set": {
                            "turma_id": nova_turma_id,
                            "data_transferencia": datetime.now()
                        }
                    },
                    session=session
                )

            return True

        except PyMongoError as e:
            print(f"Erro na transação: {e}")
            return False

🔄 Script Python para Backup Automático
import subprocess
import datetime
import os

def backup_automatico():
    """Realiza backup automático do banco de dados"""

    data_atual = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    pasta_backup = f"/backup/escola_{data_atual}"

    try:
        # Comando de backup
        comando = [
            "mongodump",
            "--host", "localhost:27017",
            "--db", "escola",
            "--gzip",
            "--out", pasta_backup
        ]

        # Executar backup
        resultado = subprocess.run(comando, capture_output=True, text=True)

        if resultado.returncode == 0:
            print(f"✅ Backup realizado com sucesso: {pasta_backup}")

            # Remover backups antigos (manter últimos 7 dias)
            limpar_backups_antigos("/backup/", dias=7)

        else:
            print(f"❌ Erro no backup: {resultado.stderr}")

    except Exception as e:
        print(f"❌ Erro ao executar backup: {e}")

def limpar_backups_antigos(pasta, dias=7):
    """Remove backups mais antigos que X dias"""
    import time

    agora = time.time()
    limite = agora - (dias * 24 * 60 * 60)

    for arquivo in os.listdir(pasta):
        caminho = os.path.join(pasta, arquivo)
        if os.path.isdir(caminho) and arquivo.startswith("escola_"):
            if os.path.getctime(caminho) < limite:
                import shutil
                shutil.rmtree(caminho)
                print(f"🗑️ Backup removido: {arquivo}")

# Agendar backup diário
if __name__ == "__main__":
    backup_automatico()

// Pipeline para relatório executivo de performance acadêmica
db.alunos.aggregate([
  // 1. Filtrar dados válidos
  {
    $match: {
      ativo: true,
      notas: { $exists: true, $ne: [] },
      data_matricula: { $gte: new Date("2024-01-01") }
    }
  },

  // 2. Adicionar campos calculados
  {
    $addFields: {
      media: { $avg: "$notas" },
      situacao: {
        $switch: {
          branches: [
            { case: { $gte: [{ $avg: "$notas" }, 9] }, then: "Excelente" },
            { case: { $gte: [{ $avg: "$notas" }, 8] }, then: "Ótimo" },
            { case: { $gte: [{ $avg: "$notas" }, 7] }, then: "Bom" },
            { case: { $gte: [{ $avg: "$notas" }, 6] }, then: "Regular" }
          ],
          default: "Insuficiente"
        }
      },
      tempo_curso: {
        $divide: [
          { $subtract: [new Date(), "$data_matricula"] },
          1000 * 60 * 60 * 24 * 30  // meses
        ]
      }
    }
  },

  // 3. Lookup com múltiplas coleções
  {
    $lookup: {
      from: "cursos",
      localField: "curso_id",
      foreignField: "_id",
      as: "curso"
    }
  },
  { $unwind: "$curso" },

  {
    $lookup: {
      from: "turmas",
      localField: "turma_id",
      foreignField: "_id",
      as: "turma"
    }
  },
  { $unwind: "$turma" },

  // 4. Agrupamento por departamento e situação
  {
    $group: {
      _id: {
        departamento: "$curso.departamento",
        situacao: "$situacao"
      },
      total_alunos: { $sum: 1 },
      media_geral: { $avg: "$media" },
      tempo_medio_curso: { $avg: "$tempo_curso" },
      alunos_detalhes: {
        $push: {
          nome: "$nome",
          curso: "$curso.nome",
          turma: "$turma.codigo",
          media: { $round: ["$media", 2] }
        }
      }
    }
  },

  // 5. Estruturar resultado final
  {
    $group: {
      _id: "$_id.departamento",
      situacoes: {
        $push: {
          situacao: "$_id.situacao",
          total: "$total_alunos",
          media: { $round: ["$media_geral", 2] },
          tempo_medio: { $round: ["$tempo_medio_curso", 1] }
        }
      },
      total_departamento: { $sum: "$total_alunos" }
    }
  },

  // 6. Ordenação final
  { $sort: { total_departamento: -1, _id: 1 } }
])

// 🎯 Resultado: Relatório completo por departamento com todas as métricas!
